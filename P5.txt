#include <avr/io.h>
#include <avr/interrupt.h>

/// Wiring
// PD6 / OC0A / D6 -> LED
// PC0 / ADC0 / A0 -> Sensor

/// Clock
// X F_CPU is defined here only to calculate the UART baud rate.
#ifndef F_CPU
#define F_CPU 16000000UL
#endif

#define BAUDRATE 9600
#define UBRR_VALUE ((F_CPU / (16UL * BAUDRATE)) - 1)

/// Shared data
// These variables are updated by the scheduler tasks.
volatile uint16_t adc_10ms = 0;
volatile uint16_t adc_sensor = 0;

// Timer1 interrupt sets this flag every 1 ms.
volatile uint8_t timer_flag = 0;

/// Timer1: 1 ms scheduler clock
void Timer1_Init(void)
{
    // Timer1 runs in CTC mode.
    // The timer counts from 0 to OCR1A, then starts again.
    TCCR1A = 0;

    // WGM12 = CTC mode
    // CS11 + CS10 = prescaler 64
    TCCR1B = (1 << WGM12) | (1 << CS11) | (1 << CS10);

    // 16 MHz / 64 = 250 kHz
    // 250 kHz / 250 = 1 kHz = 1 ms period
    OCR1A = 249;

    // Enable Timer1 Compare Match A interrupt.
    TIMSK1 = (1 << OCIE1A);
}

/// Timer0: LED PWM
void PWM_Init(void)
{
    // PD6 is OC0A, the hardware PWM output of Timer0.
    DDRD |= (1 << DDD6);

    // Fast PWM mode
    // Non-inverting output on OC0A
    TCCR0A = (1 << COM0A1) |
             (1 << WGM01) |
             (1 << WGM00);

    // Timer0 prescaler = 64
    TCCR0B = (1 << CS01) | (1 << CS00);

    // Start with 0% duty cycle.
    OCR0A = 0;
}

/// ADC: Sensor input
void ADC_Init(void)
{
    // AVcc is used as the ADC reference.
    ADMUX = (1 << REFS0);

    // Enable ADC.
    // Prescaler = 128.
    ADCSRA = (1 << ADEN) |
             (1 << ADPS2) |
             (1 << ADPS1) |
             (1 << ADPS0);
}

/// ADC: Read one conversion
uint16_t ADC_Read(uint8_t channel)
{
    // Keep the reference settings.
    // Change only the ADC channel.
    ADMUX = (ADMUX & 0xF0) | (channel & 0x0F);

    // Start ADC conversion.
    ADCSRA |= (1 << ADSC);

    // Wait until conversion is finished.
    while (ADCSRA & (1 << ADSC));

    // Return the 10-bit ADC result: 0 to 1023.
    return ADC;
}

/// UART: Initialization
void UART_Init(void)
{
    UBRR0H = (uint8_t)(UBRR_VALUE >> 8);
    UBRR0L = (uint8_t)UBRR_VALUE;

    // Enable transmitter.
    UCSR0B = (1 << TXEN0);

    // 8 data bits, 1 stop bit.
    UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
}

/// UART: Send one character
void UART_Transmit(char data)
{
    // Wait until the transmit buffer is empty.
    while (!(UCSR0A & (1 << UDRE0)));

    UDR0 = data;
}

/// UART: Send a string
void UART_SendString(const char* str)
{
    while (*str)
    {
        UART_Transmit(*str++);
    }
}

/// UART: Send an integer
void UART_SendNumber(uint16_t num)
{
    char buffer[6];
    uint8_t i = 0;

    if (num == 0)
    {
        UART_Transmit('0');
        return;
    }

    // Convert the number into decimal digits.
    while (num > 0)
    {
        buffer[i++] = (num % 10) + '0';
        num /= 10;
    }

    // Send the digits in the correct order.
    while (i > 0)
    {
        UART_Transmit(buffer[--i]);
    }
}

/// Timer1 interrupt
ISR(TIMER1_COMPA_vect)
{
    // This interrupt happens every 1 ms.
    // The main loop uses the flag as the scheduler heartbeat.
    timer_flag = 1;
}

/// Main scheduler
int main(void)
{
    Timer1_Init();
    PWM_Init();
    ADC_Init();
    UART_Init();

    // Enable global interrupts.
    sei();

    // Counters are measured in milliseconds.
    uint16_t t_10ms = 0;
    uint16_t t_20ms = 0;
    uint16_t t_100ms = 0;
    uint16_t t_1000ms = 0;

    while (1)
    {
        // Run the scheduler once for every 1 ms tick.
        if (timer_flag)
        {
            timer_flag = 0;

            t_10ms++;
            t_20ms++;
            t_100ms++;
            t_1000ms++;

            /// Task 1: Read ADC every 10 ms
            if (t_10ms >= 10)
            {
                t_10ms = 0;

                // Read ADC0 and store the result.
                adc_10ms = ADC_Read(0);
            }

            /// Task 2: Update LED/PWM every 20 ms
            if (t_20ms >= 20)
            {
                t_20ms = 0;

                // Limit the sensor value to the selected working range.
                uint16_t value = adc_sensor;

                if (value < 300)
                    value = 300;

                if (value > 555)
                    value = 555;

                // Convert ADC [300, 555]
                // to PWM [0, 255].
                uint8_t pwm = (uint32_t)(value - 300) * 255 / 255;

                // Write the duty cycle to Timer0 PWM.
                OCR0A = pwm;
            }

            /// Task 3: Read sensor every 100 ms
            if (t_100ms >= 100)
            {
                t_100ms = 0;

                // Read the sensor through ADC0.
                adc_sensor = ADC_Read(0);
            }

            /// Task 4: Send UART every 1000 ms
            if (t_1000ms >= 1000)
            {
                t_1000ms = 0;

                UART_SendString("ADC: ");
                UART_SendNumber(adc_sensor);

                UART_SendString(" | PWM: ");
                UART_SendNumber(OCR0A);

                UART_SendString("\r\n");
            }
        }
    }

    return 0;
}
