#include <Arduino.h>

#ifndef F_CPU
#define F_CPU 16000000UL
#endif

#include <avr/io.h>
#include <avr/interrupt.h>
#include <stdlib.h>

/// Wiring
// PB0 / ICP1 / D8 -> Input signal


/// X USART
void USART_Init(unsigned int baud)
{
    unsigned int ubrr = F_CPU / 16 / baud - 1;

    UBRR0H = ubrr >> 8;
    UBRR0L = ubrr;

    UCSR0B = (1 << TXEN0);
    UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
}

void USART_Transmit(char data)
{
    while (!(UCSR0A & (1 << UDRE0)));

    UDR0 = data;
}

void USART_SendString(const char* str)
{
    while (*str)
        USART_Transmit(*str++);
}

void USART_SendNum(uint32_t num)
{
    char buffer[11];

    ultoa(num, buffer, 10);
    USART_SendString(buffer);
}


/// Timer1 Input Capture
// Timer1 clock:
// 16 MHz / 8 = 2 MHz
// 1 timer tick = 0.5 us

volatile uint16_t rising_time = 0;
volatile uint16_t falling_time = 0;

volatile uint16_t pulse_width_ticks = 0;
volatile uint16_t period_ticks = 0;

volatile uint16_t previous_rising = 0;

// Measurement completed
volatile bool capture_complete = false;

// 0 = waiting for rising edge
// 1 = waiting for falling edge
volatile uint8_t edge_state = 0;


/// Timer1 + Input Capture

void Timer1_ICP1_Init()
{
    // Normal Timer1 mode
    TCCR1A = 0;
    TCCR1B = 0;

    // Input Capture configuration:
    // ICNC1 = noise canceler
    // ICES1 = capture rising edge
    // CS11  = prescaler /8
    TCCR1B =
        (1 << ICNC1) |
        (1 << ICES1) |
        (1 << CS11);

    TCNT1 = 0;

    // Clear pending Input Capture flag
    TIFR1 |= (1 << ICF1);

    // Enable Input Capture interrupt
    TIMSK1 |= (1 << ICIE1);
}


// Input Capture ISR
ISR(TIMER1_CAPT_vect)
{
    // ICR1 contains the timer value
    // captured at the signal edge.
    uint16_t capture = ICR1;

    if (edge_state == 0)
    {
        // Rising edge
        rising_time = capture;

        // Time between two rising edges = period
        if (previous_rising != 0)
        {
            period_ticks =
                rising_time - previous_rising;
        }

        previous_rising = rising_time;

        // Next capture: falling edge
        TCCR1B &= ~(1 << ICES1);

        edge_state = 1;
    }
    else
    {
        // Falling edge
        falling_time = capture;

        // Rising -> falling = HIGH pulse width
        pulse_width_ticks =
            falling_time - rising_time;

        // Next capture: rising edge
        TCCR1B |= (1 << ICES1);

        edge_state = 0;

        // Measurement is ready
        capture_complete = true;
    }
}


// Setup
void setup()
{
    // PB0 / ICP1 as input
    DDRB &= ~(1 << DDB0);

    // X USART baudrate
    USART_Init(9600);

    Timer1_ICP1_Init();

    // Enable global interrupts
    sei();
}

// Main loop
void loop()
{
    if (capture_complete)
    {
        // Safely copy data from ISR
        cli();

        uint16_t width =
            pulse_width_ticks;

        uint16_t period =
            period_ticks;

        capture_complete = false;

        sei();


        // Convert timer ticks to microseconds
        // 1 tick = 0.5 us
        uint32_t high_time_us =
            width / 2;

        uint32_t period_us =
            period / 2;


        // X Display measurement results
        USART_SendString("HIGH: ");
        USART_SendNum(high_time_us);

        USART_SendString(" us | Period: ");
        USART_SendNum(period_us);

        USART_SendString(" us\r\n");
    }
}
