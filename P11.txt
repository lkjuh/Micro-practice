#include <Arduino.h>

// =========================
// Wiring
// =========================

// A0  -> LDR analog output
// D13 -> LED
// D3-D10  -> 7-segment segments a-h
// D11, D12, A1, A2 -> digit selects

const int LIGHT_PIN = A0;
const int LED_PIN = 13;

const int SEGMENT_PINS[8] = {
    3, 4, 5, 6, 7, 8, 9, 10
};

const int DIGIT_PINS[4] = {
    11, 12, A1, A2
};

// Active LOW: LOW = ON, HIGH = OFF
const byte DIGIT_PATTERN[10] = {
    0b00111111,   // 0
    0b00000110,   // 1
    0b01011011,   // 2
    0b01001111,   // 3
    0b01100110,   // 4
    0b01101101,   // 5
    0b01111101,   // 6
    0b00000111,   // 7
    0b01111111,   // 8
    0b01101111    // 9
};

const int DARK_THRESHOLD = 400;


// =========================
// 7-Segment
// =========================

void displayNumber(int number)
{
    int digits[4] = {
        (number / 1000) % 10,
        (number / 100) % 10,
        (number / 10) % 10,
        number % 10
    };

    for (int i = 0; i < 4; i++)
    {
        // Turn all digits OFF
        for (int j = 0; j < 4; j++)
            digitalWrite(DIGIT_PINS[j], HIGH);

        // Set segment pattern
        for (int j = 0; j < 8; j++)
        {
            digitalWrite(
                SEGMENT_PINS[j],
                bitRead(DIGIT_PATTERN[digits[i]], j) ? LOW : HIGH
            );
        }

        // Turn current digit ON
        digitalWrite(DIGIT_PINS[i], LOW);

        // X: Multiplexing refresh delay
        delay(2);
    }
}


// =========================
// Setup
// =========================

void setup()
{
    pinMode(LIGHT_PIN, INPUT);
    pinMode(LED_PIN, OUTPUT);

    // 7-segment pins
    for (int i = 0; i < 8; i++)
    {
        pinMode(SEGMENT_PINS[i], OUTPUT);
        digitalWrite(SEGMENT_PINS[i], HIGH);
    }

    // Digit select pins
    for (int i = 0; i < 4; i++)
    {
        pinMode(DIGIT_PINS[i], OUTPUT);
        digitalWrite(DIGIT_PINS[i], HIGH);
    }

    Serial.begin(9600);
}


// =========================
// Main Loop
// =========================

void loop()
{
    // Read LDR using the ATmega328 ADC
    int adcValue = analogRead(LIGHT_PIN);

    // Dark / Bright detection
    if (adcValue < DARK_THRESHOLD)
    {
        digitalWrite(LED_PIN, HIGH);
    }
    else
    {
        digitalWrite(LED_PIN, LOW);
    }

    // UART
    static unsigned long lastSerial = 0;

    if (millis() - lastSerial >= 100)
    {
        lastSerial = millis();

        Serial.print("ADC: ");
        Serial.print(adcValue);

        if (adcValue < DARK_THRESHOLD)
            Serial.println(" - Dark");
        else
            Serial.println(" - Bright");
    }

    // 7-segment display
    displayNumber(adcValue);
}
