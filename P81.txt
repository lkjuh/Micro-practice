#include <avr/io.h>
#include <util/delay.h>

/// Wiring
// X LED display only. No external device.
// PB0 -> LED0
// PB1 -> LED1
// PD2-PD7 -> LED2-LED7


// I2C Initialization
void i2c_init(void)
{
    TWSR = 0x00;                  // Prescaler = 1
    TWBR = 72;                    // ~100 kHz I2C clock at 16 MHz
    TWCR = (1 << TWEN);           // Enable TWI hardware
}


// I2C START
void i2c_start(void)
{
    // Request the TWI hardware to generate START
    TWCR = (1 << TWINT) |
           (1 << TWSTA) |
           (1 << TWEN);

    // Wait until the START operation is completed
    while (!(TWCR & (1 << TWINT)));
}


// I2C WRITE
void i2c_write(uint8_t data)
{
    // X Display the transmitted byte on LEDs
    // X Lower 2 bits -> PB0 and PB1
    PORTB = (PORTB & 0xFC) | (data & 0x03);

    // X Upper 6 bits -> PD2 to PD7
    PORTD = (PORTD & 0x03) | (data & 0xFC);

    // Put the byte into the TWI data register
    TWDR = data;

    // Start transmitting the byte
    TWCR = (1 << TWINT) |
           (1 << TWEN);

    // Wait until the transmission is completed
    while (!(TWCR & (1 << TWINT)));
}

// I2C READ WITH ACK
uint8_t i2c_read_ack(void)
{
    // Enable ACK after receiving a byte
    TWCR = (1 << TWINT) |
           (1 << TWEN) |
           (1 << TWEA);

    // Wait until the byte is received
    while (!(TWCR & (1 << TWINT)));

    // Return the received byte
    return TWDR;
}


// I2C READ WITH NACK
uint8_t i2c_read_nack(void)
{
    // Do not enable TWEA
    // NACK indicates the final byte
    TWCR = (1 << TWINT) |
           (1 << TWEN);

    // Wait until the byte is received
    while (!(TWCR & (1 << TWINT)));

    // Return the received byte
    return TWDR;
}


// I2C STOP
void i2c_stop(void)
{
    // Request the TWI hardware to generate STOP
    TWCR = (1 << TWINT) |
           (1 << TWSTO) |
           (1 << TWEN);
}


// Main
int main(void)
{
    // X Configure LED pins as outputs
    DDRB |= 0b00000011;           // PB0, PB1
    DDRD |= 0b11111100;           // PD2-PD7

    // Initialize I2C
    i2c_init();


    /// X LED Demo Data ouput
    uint8_t sample_data[] = {
        0x75,
        0x6B,
        0x00,
        0x3B,
        0x1C,
        0x08,
        0x43,
        0x1A
    };

    // X Number of LED display states
    uint8_t total_patterns =
        sizeof(sample_data) / sizeof(sample_data[0]);

    // X Start from the first display state
    uint8_t current_index = 0;


    while (1)
    {
        // X Send the current byte so it is displayed on LEDs
        i2c_start();

        // X Example I2C device address (write)
        i2c_write((0x68 << 1) | 0);

        // X Send the current demo byte
        i2c_write(sample_data[current_index]);

        i2c_stop();


        // X Move to the next LED display state
        current_index++;

        // X Return to the first state after the last one
        if (current_index >= total_patterns)
            current_index = 0;

        // X Keep each LED state for 500 ms
        _delay_ms(500);
    }

    return 0;
}
