//Setup
//PB0-1 -> Led0-1
//PD2-7 -> Led2-7
//PC0-3 -> Button 0-3
//PB2-5 -> Button 4-7

// No external device
// Data out show by led
// Input in by button with capturing mode (Buffer show by led)

#include <avr/io.h>
#include <util/delay.h>
#include <stdlib.h> // Required for rand()

// --- I2C FUNCTIONS ---
void i2c_init(void) {
    TWSR = 0x00; 
    TWBR = 72;   
    TWCR = (1 << TWEN); 
}

void i2c_start(void) {
    TWCR = (1 << TWINT) | (1 << TWSTA) | (1 << TWEN);
    // while (!(TWCR & (1 << TWINT))); // Commented out for simulation
}

void i2c_write(uint8_t data) {
    TWDR = data;
    TWCR = (1 << TWINT) | (1 << TWEN);
    // while (!(TWCR & (1 << TWINT))); // Commented out for simulation
}

void i2c_stop(void) {
    TWCR = (1 << TWINT) | (1 << TWSTO) | (1 << TWEN);
}

// --- HARDWARE HELPER FUNCTIONS ---
// Output bits 0-1 to PB0-PB1, and bits 2-7 to PD2-PD7
void display_byte(uint8_t val) {
    PORTB = (PORTB & 0xFC) | (val & 0x03);
    PORTD = (PORTD & 0x03) | (val & 0xFC);
}

// Read Buttons 0-3 from PC0-PC3 and Buttons 4-7 from PB2-PB5
uint8_t read_buttons(void) {
    uint8_t c_val = (~PINC) & 0x0F; // Read PC0-PC3 (Bits 0-3)
    uint8_t b_val = (~PINB) & 0x3C; // Read PB2-PB5 (Bits 2-5)
    
    uint8_t current = c_val;
    current |= (b_val << 2);        // Shift PB2-PB5 up to Bits 4-7
    return current;
}

// Rotates 8-bit value to the left by shift_amount (1 to 4 bits)
uint8_t rotate_left(uint8_t val, uint8_t shift) {
    return (val << shift) | (val >> (8 - shift));
}

int main(void) {
    // 1. Configure LED Pins as Outputs (PB0-PB1 and PD2-PD7)
    DDRB |= 0b00000011; 
    DDRD |= 0b11111100; 

    // 2. Configure Button Pins as Inputs with Pull-ups (PC0-PC3 and PB2-PB5)
    DDRC &= ~0x0F;                      
    PORTC |= 0x0F; // Enable pull-ups PC0-PC3
    DDRB &= ~0x3C; 
    PORTB |= 0x3C; // Enable pull-ups PB2-PB5

    i2c_init();

    uint8_t packet_data = 0x01; // Initial default data packet
    uint16_t seed_counter = 0;  // Entropy counter for PRNG seed

    while (1) {
        // =================================================================
        // PHASE 1: TRANSMIT & DISPLAY DATA (SHIFTED RESULT)
        // =================================================================
        display_byte(packet_data);

        i2c_start();
        i2c_write((0x68 << 1) | 0);
        i2c_write(packet_data);
        i2c_stop();

        _delay_ms(2000); // Hold display for 2s while ignoring buttons

        // =================================================================
        // PHASE 2: INPUT CAPTURE
        // =================================================================
        // 1. Begin Flash Signal (0.5s ON, then OFF)
        display_byte(0xFF);
        _delay_ms(500);
        display_byte(0x00);

        // Flush any held button presses from transition
        while (read_buttons() > 0) {
            _delay_ms(10);
            seed_counter++; // Collect timing entropy
        }

        uint8_t input_buffer = 0x00;
        uint8_t last_buttons = 0x00;
        uint8_t timer_active = 0;      
        uint16_t inactivity_ticks = 0; // 2s countdown (40 * 50ms)
        uint16_t total_ticks = 0;      // 10s global timeout (200 * 50ms)
        uint8_t capturing = 1;

        // 2. Input Capture Loop
        while (capturing) {
            uint8_t current_buttons = read_buttons();
            uint8_t newly_pressed = current_buttons & ~last_buttons;

            if (newly_pressed > 0) {
                if (!timer_active) {
                    timer_active = 1; // Start 2s inactivity timer on FIRST click
                }
                input_buffer |= newly_pressed;
                inactivity_ticks = 0; // Reset 2s timer on every click
            }

            last_buttons = current_buttons;

            // Display live input buffer on LEDs (unshifted)
            display_byte(input_buffer);

            total_ticks++;
            seed_counter++; // Continuously increment seed based on user response time
            
            if (timer_active) {
                inactivity_ticks++;
            }

            // Exit conditions: 2s inactivity after first press OR 10s timeout
            if ((timer_active && inactivity_ticks >= 40) || (total_ticks >= 200)) {
                capturing = 0;
            }

            _delay_ms(50);
        }

        // 3. End Flash Signal (0.5s ON, then OFF)
        display_byte(0xFF);
        _delay_ms(500);
        display_byte(0x00);

        // Seed random generator with execution entropy
        srand(seed_counter);

        // Calculate random shift amount from 1 to 4 bits
        uint8_t shift_amount = 1;

        // Apply circular bit shift to input_buffer before sending
        packet_data = rotate_left(input_buffer, shift_amount);
    }
}
