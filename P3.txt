#ifndef F_CPU
#define F_CPU 16000000UL
#endif

#include <Arduino.h>
#include <avr/io.h>
#include <util/delay.h>
#include <LiquidCrystal.h>

// Wiring:
// PB1 / D9  -> Servo signal
// PB3 / D11 -> LED
// LCD RS    -> D12
// LCD EN    -> D11
// LCD D4    -> D4
// LCD D5    -> D5
// LCD D6    -> D6
// LCD D7    -> D7

// X LCD: used only for displaying angle and PWM value
LiquidCrystal lcd(12, 11, 4, 5, 6, 7);

#define SERVO_PIN PB1
#define LED_PIN   PB3

/// X LCD

void LCD_PrintStatus(uint8_t angle)
{
    uint8_t pwm_val = OCR2A;

    lcd.setCursor(0, 0);
    lcd.print("SERVO & LED CTRL");

    lcd.setCursor(0, 1);
    lcd.print("Goc:");
    lcd.print(angle);
    lcd.print(" PWM:");
    lcd.print(pwm_val);
    lcd.print("    ");
}

/// Timer1 - Servo PWM 50 Hz

void Servo_PWM_Init(void)
{
    // PB1 / OC1A -> Servo output
    DDRB |= (1 << SERVO_PIN);

    // Fast PWM Mode 14, non-inverting output
    TCCR1A = (1 << COM1A1) | (1 << WGM11);
    TCCR1B = (1 << WGM13) | (1 << WGM12) | (1 << CS11);

    // 16 MHz / 8 = 2 MHz
    // 20 ms period -> 40000 timer ticks
    ICR1 = 39999;
}

void Servo_SetAngle(uint8_t angle)
{
    if (angle > 180)
        angle = 180;

    uint16_t min_ocr = 999;   // 0.5 ms
    uint16_t max_ocr = 4999;  // 2.5 ms

    OCR1A = min_ocr
          + ((uint32_t)angle * (max_ocr - min_ocr)) / 180;
}

/// Timer2 - LED PWM

void LED_PWM_Init(void)
{
    // PB3 / OC2A -> LED output
    DDRB |= (1 << LED_PIN);

    // Fast PWM 8-bit, non-inverting output
    TCCR2A = (1 << WGM20) | (1 << WGM21) | (1 << COM2A1);

    // Prescaler = 64
    TCCR2B = (1 << CS22);
}

void LED_SetBrightnessByAngle(uint8_t angle)
{
    if (angle > 180)
        angle = 180;

    uint8_t pwm_val =
        (uint16_t)angle * 255 / 180;

    OCR2A = pwm_val;
}

/// Main

int main(void)
{
    // X LCD initialization
    lcd.begin(16, 2);
    lcd.clear();

    Servo_PWM_Init();
    LED_PWM_Init();

    while (1)
    {
        // Sweep from 0 to 180 degrees
        for (uint8_t angle = 0; angle <= 180; angle += 5)
        {
            Servo_SetAngle(angle);
            LED_SetBrightnessByAngle(angle);

            // X Display current angle and LED PWM
            LCD_PrintStatus(angle);

            _delay_ms(50);
        }

        _delay_ms(500);

        // Sweep from 180 back to 0 degrees
        for (int16_t angle = 180; angle >= 0; angle -= 5)
        {
            Servo_SetAngle((uint8_t)angle);
            LED_SetBrightnessByAngle((uint8_t)angle);

            // X Display current angle and LED PWM
            LCD_PrintStatus((uint8_t)angle);

            _delay_ms(50);
        }

        _delay_ms(500);
    }

    return 0;
}
